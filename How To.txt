#How to install bettercap

#Updating your machine
sudo apt update && sudo apt full-upgrade -y

#Checking if Go is installed
go version

#If Go is version 1.19 or lower, time for an update
sudo apt remove golang-go -y
sudo apt install golang-go -y

#If you never had go before, Install go
sudo apt install golang-go -y

#Installing prerequisites
sudo apt install -y build-essential pkg-config libpcap-dev libusb-1.0-0-dev libnetfilter-queue-dev

#Installing Bettercap using Go | Run the one on line 21 for the latest version of bettercap | Do not run both line 20 and 21, pick one
#go install github.com/bettercap/bettercap@latest
go install github.com/bettercap/bettercap/v2@latest

#Put it on path
sudo mv ~/go/bin/bettercap /usr/local/bin/

#Verify installation
bettercap -h

#Install additional modules (caplets & web assets)
sudo mkdir -p /usr/local/share/bettercap
cd /usr/local/share/bettercap
sudo git clone https://github.com/bettercap/caplets
